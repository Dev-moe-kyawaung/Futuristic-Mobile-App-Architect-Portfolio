// Dynamic feature module configuration
// feature:onboarding/build.gradle.kts
plugins {
  id("com.android.dynamic-feature")
  kotlin("android")
}

android {
  namespace = "com.example.app.feature.onboarding"
}

dependencies {
  implementation(project(":app"))
  implementation(project(":core:ui"))
  implementation(project(":core:network"))
}

// Installation with Play Core
class FeatureInstaller @Inject constructor(
  private val context: Context
) {
  private val splitInstallManager = SplitInstallManagerFactory.getInstance(context)
  
  suspend fun installOnboardingFeature(): Result {
    val request = SplitInstallRequest.newBuilder()
      .addFeature("onboarding")
      .build()
    
    return try {
      splitInstallManager.startInstall(request).await()
      Result.success()
    } catch (e: Exception) {
      Result.failure(e)
    }
  }
}
