// App Startup initializers
class CrashReportingInitializer : Initializer<Crashlytics> {
  override fun create(context: Context): Crashlytics {
    return FirebaseCrashlytics.getInstance()
  }
  
  override fun dependencies(): List<Class<out Initializer<*>>> {
    return emptyList() // No dependencies
  }
}

class AnalyticsInitializer : Initializer<AnalyticsTracker> {
  override fun create(context: Context): AnalyticsTracker {
    return AnalyticsTrackerImpl(context)
  }
  
  override fun dependencies(): List<Class<out Initializer<*>>> {
    return listOf(CrashReportingInitializer::class.java)
  }
}
