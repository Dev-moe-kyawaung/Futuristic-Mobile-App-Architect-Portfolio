// Crashlytics integration
class CrashReportingTree : Tree() {
  override fun log(priority: Int, tag: String?, message: String, t: Throwable?) {
    if (priority == Log.ERROR || t != null) {
      FirebaseCrashlytics.getInstance().recordException(t ?: Exception(message))
    }
  }
}

// Custom logging
object AppLogger {
  private const val TAG = "MyApp"
  
  fun d(message: String) {
    Log.d(TAG, message)
    if (BuildConfig.DEBUG) {
      Timber.d(message)
    }
  }
  
  fun e(message: String, throwable: Throwable? = null) {
    Log.e(TAG, message, throwable)
    FirebaseCrashlytics.getInstance().recordException(throwable ?: Exception(message))
  }
}

// Performance monitoring
fun trackPerformance(operation: String, block: () -> Unit) {
  val monitor = FirebasePerformance.getInstance().newTrace(operation)
  monitor.start()
  try {
    block()
  } finally {
    monitor.stop()
  }
}
