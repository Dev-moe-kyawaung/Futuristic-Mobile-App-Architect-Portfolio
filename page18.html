// Feature flag manager
class FeatureFlagManager @Inject constructor(
  private val remoteConfig: FirebaseRemoteConfig
) {
  val isNewOnboardingEnabled: Boolean
    get() = remoteConfig.getBoolean("feature_new_onboarding")
  
  val buttonColor: Int
    get() = remoteConfig.getLong("ui_button_color").toInt()
  
  suspend fun fetchAndActivate() {
    remoteConfig.fetchAndActivate()
  }
}

// A/B test integration
class ExperimentTracker @Inject constructor(
  private val abTesting: FirebaseABTesting
) {
  fun startOnboardingTest(): Boolean {
    val experiment = abTesting.currentExperiment
    return experiment?.experimentId == "onboarding_redesign_v2"
  }
  
  fun trackConversion(event: String) {
    FirebaseAnalytics.getInstance().logEvent(event, null)
  }
}
