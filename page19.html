Design System Tokens

Centralized design tokens for colors, spacing, typography, and motion.

Token Architecture

🎨 Color Tokens

Primary, secondary, semantic colors

📏 Spacing Scale

4px base grid system

📝 Typography

Font families, weights, sizes

⚡ Motion

Duration, easing curves

// Design tokens object
object DesignTokens {
  object Colors {
    val Primary = Color(0xFF00F3FF)
    val Secondary = Color(0xFFFF2A9F)
    val Success = Color(0xFF00C853)
    val Error = Color(0xFFD50000)
    val Warning = Color(0xFFFFAB00)
  }
  
  object Spacing {
    val Xs = 4.dp
    val Sm = 8.dp
    val Md = 16.dp
    val Lg = 24.dp
    val Xl = 32.dp
    val Xxl = 48.dp
  }
  
  object Typography {
    val H1 = TextStyle(
      fontSize = 32.sp,
      fontWeight = FontWeight.Bold,
      lineHeight = 40.sp
    )
    val H2 = TextStyle(
      fontSize = 24.sp,
      fontWeight = FontWeight.SemiBold,
      lineHeight = 32.sp
    )
    val Body = TextStyle(
      fontSize = 16.sp,
      fontWeight = FontWeight.Normal,
      lineHeight = 24.sp
    )
  }
  
  object Motion {
    val DurationShort = 150.milliseconds
    val DurationMedium = 300.milliseconds
    val DurationLong = 500.milliseconds
    
    val EasingStandard = FastOutSlowInEasing
    val EasingBouncy = Spring.DampingRatioMediumBouncy
  }
}
