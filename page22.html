// Repository with offline-first pattern
class UserRepositoryImpl @Inject constructor(
  private val userDao: UserDao,
  private val userApi: UserApi,
  private val networkMonitor: NetworkMonitor
) : UserRepository {
  
  override fun getUserStream(userId: String): Flow<User?> {
    return userDao.getUserFlow(userId)
  }
  
  override suspend fun refreshUser(userId: String): Result<User> {
    return try {
      val remoteUser = userApi.getUser(userId)
      userDao.upsertUser(remoteUser.toEntity())
      Result.success(remoteUser)
    } catch (e: IOException) {
      // Network error, return cached data
      val cached = userDao.getUser(userId)
      if (cached != null) {
        Result.success(cached.toDomain())
      } else {
        Result.failure(e)
      }
    }
  }
  
  override suspend fun updateUser(user: User): Result<Unit> {
    // Optimistic update
    userDao.upsertUser(user.toEntity())
    
    return try {
      userApi.updateUser(user.toDto())
      Result.success(Unit)
    } catch (e: Exception) {
      // Rollback on failure
      userDao.deleteUser(user.id)
      Result.failure(e)
    }
  }
}

// WorkManager for background sync
class SyncWorker(
  context: Context,
  params: WorkerParameters
) : CoroutineWorker(context, params) {
  
  override suspend fun doWork(): Result {
    val repository = getKoin().get<DataRepository>()
    
    return try {
      repository.syncAllData()
      Result.success()
    } catch (e: Exception) {
      if (runAttemptCount < 3) Result.retry()
      else Result.failure()
    }
  }
}
