// Play Store deployment tracks:
// 1. Internal Testing (immediate)
// 2. Closed Testing (beta testers)
// 3. Open Testing (public beta)
// 4. Production (staged rollout)

// Gradle Play Publisher configuration
play {
  serviceAccountCredentials.set(file("play-key.json"))
  track.set("internal")
  releaseStatus.set(com.github.triplet.gradle.androidpublisher.ReleaseStatus.COMPLETED)
  
  defaultToAppBundles.set(true)
  commit.set(true)
  
  // Staged rollout percentage
  userFraction.set(0.1) // 10% initially
}

// Release naming convention
// {versionName}-{buildType}-{track}
// 2.4.1-internal, 2.4.1-beta, 2.4.1-production

// Automated release notes
tasks.register("generateReleaseNotes") {
  doLast {
    val lastTag = "git describe --tags --abbrev=0".execute().text.trim()
    val commits = "git log --oneline $lastTag..HEAD".execute().text
    
    file("release-notes.txt").writeText(
      """
      ## What's New
      
      $commits
      
      ## Technical Changes
      
      - Dependency updates
      - Performance improvements
      - Bug fixes
      """.trimIndent()
    )
  }
}
