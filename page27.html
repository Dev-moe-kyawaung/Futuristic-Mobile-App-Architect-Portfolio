// Billing integration
class BillingManager @Inject constructor(
  private val context: Context
) {
  private val billingClient = BillingClient.newBuilder(context)
    .setUpdatedConnectionListener { isReady ->
      if (isReady) queryPurchases()
    }
    .build()
  
  fun startConnection() {
    billingClient.startConnection()
  }
  
  fun queryPurchases() {
    billingClient.queryPurchasesAsync(
      QueryPurchasesParams.newBuilder()
        .setProductType(BillingClient.ProductType.SUBS)
        .build()
    ) { billingResult, purchases ->
      if (billingResult.isOk && purchases != null) {
        // Process active subscriptions
        handlePurchases(purchases)
      }
    }
  }
  
  fun launchBillingFlow(activity: Activity, productId: String) {
    val flowParams = BillingFlowParams.newBuilder()
      .setProductDetailsParamsList(
        listOf(
          ProductDetailsParams.newBuilder()
            .setProductDetails(getProductDetails(productId))
            .build()
        )
      )
      .build()
    
    billingClient.launchBillingFlow(activity, flowParams)
  }
}

// Analytics event tracking
object Analytics {
  fun trackScreenView(screenName: String) {
    FirebaseAnalytics.getInstance().logEvent(
      "screen_view",
      bundle {
        param(FirebaseAnalytics.Param.SCREEN_NAME, screenName)
      }
    )
  }
  
  fun trackPurchase(productId: String, revenue: Double) {
    FirebaseAnalytics.getInstance().logEvent(
      "purchase",
      bundle {
        param(FirebaseAnalytics.Param.ITEM_ID, productId)
        param(FirebaseAnalytics.Param.VALUE, revenue)
      }
    )
  }
}
