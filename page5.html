// Domain Layer - Use Case
class GetUserProfileUseCase(
  private val userRepository: UserRepository
) {
  suspend operator fun invoke(userId: String): Result<UserProfile> {
    return userRepository.getUserProfile(userId)
      .map { it.toDomainModel() }
  }
}

// Data Layer - Repository Implementation
class UserRepositoryImpl(
  private val userDao: UserDao,
  private val userApi: UserApi
) : UserRepository {
  override suspend fun getUserProfile(userId: String): Result<UserEntity> {
    return try {
      val cached = userDao.getUser(userId)
      if (cached != null) {
        Result.success(cached)
      } else {
        val remote = userApi.fetchUser(userId)
        userDao.insertUser(remote.toEntity())
        Result.success(remote.toEntity())
      }
    } catch (e: Exception) {
      Result.failure(e)
    }
  }
}

// Presentation Layer - ViewModel
@HiltViewModel
class ProfileViewModel @Inject constructor(
  private val getUserProfile: GetUserProfileUseCase
) : ViewModel() {
  private val _uiState = MutableStateFlow<ProfileUiState>(ProfileUiState.Loading)
  val uiState: StateFlow<ProfileUiState> = _uiState.asStateFlow()

  init {
    loadUserProfile()
  }

  private fun loadUserProfile() {
    viewModelScope.launch {
      _uiState.value = ProfileUiState.Loading
      getUserProfile("user123")
        .onSuccess { profile ->
          _uiState.value = ProfileUiState.Success(profile.toUiModel())
        }
        .onFailure { error ->
          _uiState.value = ProfileUiState.Error(error.message)
        }
    }
  }
}

Modularization Strategy

Multi-module architecture for improved build times, team scalability, and feature isolation.

📦 :app Module

Application entry point, dependency injection setup, and app-level configuration. Minimal code, primarily wiring.

🎨 :feature:* Modules

Self-contained feature modules (home, profile, settings) with their own UI, ViewModel, and navigation graphs.

🧩 :core:* Modules

Shared utilities: network, database, UI components, analytics, and common business logic.

🔌 :dynamic-feature Modules

On-demand features delivered via Play Feature Delivery for reduced initial APK size.

📊 :benchmark Module

Isolated performance testing module for startup, memory, and rendering benchmarks.

🧪 :testing Module

Shared test utilities, mocks, and testing infrastructure across all modules.

Dependency Injection Architecture

Hilt for compile-time DI with clear scoping and module organization across the multi-module structure.

🏗️ AppModule

  • @Singleton scope
  • Retrofit instance
  • Database instance
  • Analytics tracker
  • App-wide dependencies

📦 Feature Modules

  • @ViewModelScoped
  • Use case bindings
  • Repository implementations
  • Feature-specific deps
  • Isolated from other features

🧩 Core Modules

  • @InstallIn(SingletonComponent::class)
  • Network module
  • Database module
  • UI component providers

// :core:network - NetworkModule.kt
@Module
@InstallIn(SingletonComponent::class)
object NetworkModule {

  @Provides
  @Singleton
  fun provideRetrofit(
    @ApplicationContext context: Context,
    okHttpClient: OkHttpClient
  ): Retrofit {
    return Retrofit.Builder()
      .baseUrl("https://api.example.com/")
      .client(okHttpClient)
      .addConverterFactory(MoshiConverterFactory.create())
      .build()
  }

  @Provides
  @Singleton
  fun provideOkHttpClient(
    logger: HttpLoggingInterceptor
  ): OkHttpClient {
    return OkHttpClient.Builder()
      .addInterceptor(logger)
      .addInterceptor(ConnectivityInterceptor())
      .connectTimeout(30, TimeUnit.SECONDS)
      .readTimeout(30, TimeUnit.SECONDS)
      .build()
  }

  @Provides
  @Singleton
  fun provideApiService(retrofit: Retrofit): UserApi {
    return retrofit.create(UserApi::class.java)
  }
}

State Management Patterns

Unidirectional data flow with StateFlow, sealed classes for UI state, and event handling for one-time actions.

🔀 MVI Pattern

Model-View-Intent: User actions (Intents) update state through ViewModels, UI reacts to state changes.

📊 StateFlow

Hot streams for UI state, lifecycle-aware, composable with coroutines, and testable.

🎯 Sealed UI States

Type-safe state representation: Loading, Success(data), Error(message), Empty.

⚡ Channel for Events

One-time events (navigation, snacks) via Channel or SharedFlow with extraBufferCapacity.

// UI State definition
sealed class ProfileUiState {
  object Loading : ProfileUiState()
  data class Success(val profile: ProfileUiModel) : ProfileUiState()
  data class Error(val message: String) : ProfileUiState()
  object Empty : ProfileUiState()
}

// Event handling with Channel
class ProfileViewModel @Inject constructor(
  private val getUserProfile: GetUserProfileUseCase,
  private val updateProfile: UpdateProfileUseCase
) : ViewModel() {
  
  private val _uiState = MutableStateFlow<ProfileUiState>(ProfileUiState.Loading)
  val uiState: StateFlow<ProfileUiState> = _uiState.asStateFlow()

  private val _events = Channel<ProfileEvent>()
  val events: Flow<ProfileEvent> = _events.receiveAsFlow()

  sealed class ProfileEvent {
    data class NavigateToEdit(val userId: String) : ProfileEvent()
    data class ShowSnackbar(val message: String) : ProfileEvent()
    object NavigateBack : ProfileEvent()
  }

  fun onIntent(intent: ProfileIntent) {
    when (intent) {
      is ProfileIntent.LoadProfile -> loadProfile(intent.userId)
      is ProfileIntent.UpdateProfile -> updateProfile(intent.userId, intent.data)
      is ProfileIntent.EditProfile -> navigateToEdit(intent.userId)
    }
  }

  private fun loadProfile(userId: String) {
    viewModelScope.launch {
      _uiState.value = ProfileUiState.Loading
      getUserProfile(userId)
        .onSuccess { profile ->
          _uiState.value = ProfileUiState.Success(profile.toUiModel())
        }
        .onFailure { error ->
          _uiState.value = ProfileUiState.Error(error.message ?: "Unknown error")
        }
    }
  }

  private suspend fun navigateToEdit(userId: String) {
    _events.send(ProfileEvent.NavigateToEdit(userId))
  }
}

// Compose UI consumption
@Composable
fun ProfileScreen(viewModel: ProfileViewModel = hiltViewModel()) {
  val uiState by viewModel.uiState.collectAsState()
  val scope = rememberCoroutineScope()

  LaunchedEffect(Unit) {
    viewModel.events.collect { event ->
      when (event) {
        is ProfileEvent.NavigateToEdit -> navController.navigate("edit/${event.userId}")
        is ProfileEvent.ShowSnackbar -> snackbarHost.showSnackbar(event.message)
        ProfileEvent.NavigateBack -> navController.popBackStack()
      }
    }
  }

  when (val state = uiState) {
    is ProfileUiState.Loading -> LoadingIndicator()
    is ProfileUiState.Success -> ProfileContent(state.profile)
    is ProfileUiState.Error -> ErrorView(state.message)
    ProfileUiState.Empty -> EmptyStateView()
  }
}
