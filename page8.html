// Unit Test Example - ViewModel with MockK and Turbine
@Test
fun `given valid user id, when load profile, then success state emitted`() = runTest {
  // Given
  val mockUseCase = mockk<GetUserProfileUseCase>()
  val expectedProfile = UserProfile(id = "123", name = "John Doe")
  
  coEvery { mockUseCase("123") } returns Result.success(expectedProfile)
  
  val viewModel = ProfileViewModel(mockUseCase)
  
  // When
  val states = viewModel.uiState.toList()
  
  // Then
  assertEquals(2, states.size)
  assertTrue(states[0] is ProfileUiState.Loading)
  assertTrue(states[1] is ProfileUiState.Success)
  assertEquals(expectedProfile, (states[1] as ProfileUiState.Success).profile)
}

// Integration Test - Room Database
@RunWith(AndroidJUnit4::class)
class UserDaoTest {
  
  @get:Rule
  val dbTestRule = InstantTaskExecutorRule()
  
  private lateinit var database: AppDatabase
  private lateinit var userDao: UserDao
  
  @Before
  fun setup() {
    database = Room.inMemoryDatabaseBuilder(
      InstrumentationRegistry.getInstrumentation().targetContext,
      AppDatabase::class.java
    ).build()
    userDao = database.userDao()
  }
  
  @After
  fun teardown() {
    database.close()
  }
  
  @Test
  fun insertAndRetrieveUser() = runBlocking {
    // Given
    val user = UserEntity(id = "123", name = "Test User")
    
    // When
    userDao.insertUser(user)
    val retrieved = userDao.getUserById("123")
    
    // Then
    assertNotNull(retrieved)
    assertEquals(user.id, retrieved?.id)
    assertEquals(user.name, retrieved?.name)
  }
}
